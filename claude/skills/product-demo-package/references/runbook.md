**Goal:** Regenerate the full AFG/Alex deliverables package — a branded Remotion product-demo video, executive System Documentation (PDF), a Centrex→Salesforce Migration Plan (PDF), and a Meeting Strategy (PDF) — and deliver them to Telegram, from a clean checkout with zero prior context.

**Architecture:** Three independent executive docs are generated by **parallel agents** (one Markdown file each), then rendered to brand-styled PDFs with a headless-Chromium `page.pdf()` pipeline. The demo video is a **hybrid**: a real Playwright screen-capture of the running app (seeded so it needs no live Centrex) composited in **Remotion** with a branded intro/outro, timed section captions, a watermark, and a rasterized-PDF overlay (because Playwright's Chromium has no PDF viewer). All heavy tooling runs inside the **Playwright Docker image** (it already bundles Node, Chromium, xvfb, poppler-installable, and a full static ffmpeg is `npm i`-able), so nothing is installed on the host except the Remotion CLI/MCP.

**Tech Stack:** Remotion 4 (`@remotion/cli`, `@remotion/mcp`), React 18, Playwright 1.59 (`mcr.microsoft.com/playwright:v1.59.1-noble`), `marked` (MD→HTML), `@ffmpeg-installer/ffmpeg`, `poppler-utils` (`pdftoppm`), Laravel 13 app (Docker Compose, port 8088), OpenAI/Anthropic classifier (already configured in `dashboard/.env`).

```jsx
import React from 'react';
import {
  AbsoluteFill, Sequence, OffthreadVideo, staticFile,
  useCurrentFrame, useVideoConfig, interpolate, spring,
} from 'remotion';

const NAVY = '#0A1628';
const NAVY700 = '#1A2742';
const ACCENT = '#3B82F6';
const FONT = "'Figtree','Segoe UI',Helvetica,Arial,ui-sans-serif,system-ui,sans-serif";

// Lower-third caption windows, in seconds within the captured footage (set to Task 6 scenes).
const CAPTIONS = [
  { t: 0, d: 10, text: 'Secure login — password or one-tap magic link' },
  { t: 10, d: 14, text: 'Live deal dashboard — pipeline, KPIs & decline reasons at a glance' },
  { t: 24, d: 30, text: 'Filterable deal queue — every submission tracked from one screen' },
  { t: 54, d: 15, text: 'Deal detail: merchant profile, detected positions & one-click AI Analyze' },
  { t: 69, d: 13, text: 'Split-view: transactions beside the original statement — auto-reconciled, and flagged when they don’t' },
  { t: 82, d: 17, text: 'Integrity findings & agent productivity — every decision audited' },
];

// The captured left pane is blank (Playwright's Chromium has no PDF viewer), so the real
// rasterized statement page is composited over it during the split-view window.
const PdfOverlay = ({ durationInFrames }) => {
  const frame = useCurrentFrame();
  const op = interpolate(frame, [0, 10, durationInFrames - 10, durationInFrames], [0, 1, 1, 0], { extrapolateLeft: 'clamp', extrapolateRight: 'clamp' });
  return (
    <AbsoluteFill style={{ opacity: op }}>
      <div style={{ position: 'absolute', left: 82, top: 90, width: 636, height: 772, background: '#f1f5f9', display: 'flex', alignItems: 'center', justifyContent: 'center', overflow: 'hidden' }}>
        <img src={staticFile('statement-1.png')} style={{ maxWidth: '95%', maxHeight: '97%', objectFit: 'contain', boxShadow: '0 6px 22px rgba(0,0,0,0.18)' }} />
      </div>
    </AbsoluteFill>
  );
};

const Intro = ({ durationInFrames }) => {
  const frame = useCurrentFrame();
  const { fps } = useVideoConfig();
  const rise = spring({ frame, fps, config: { damping: 200 }, durationInFrames: 40 });
  const y = interpolate(rise, [0, 1], [40, 0]);
  const op = interpolate(frame, [0, 18], [0, 1], { extrapolateRight: 'clamp' });
  const out = interpolate(frame, [durationInFrames - 12, durationInFrames], [1, 0], { extrapolateLeft: 'clamp' });
  return (
    <AbsoluteFill style={{ background: `radial-gradient(120% 120% at 50% 0%, ${NAVY700} 0%, ${NAVY} 60%)`, justifyContent: 'center', alignItems: 'center', fontFamily: FONT }}>
      <div style={{ opacity: op * out, transform: `translateY(${y}px)`, textAlign: 'center', color: '#fff' }}>
        <div style={{ color: ACCENT, letterSpacing: 6, fontSize: 20, fontWeight: 700, textTransform: 'uppercase', marginBottom: 18 }}>AFG · Alternative Funding Group</div>
        <div style={{ fontSize: 92, fontWeight: 800, letterSpacing: -2, lineHeight: 1 }}>MCA-Tracker</div>
        <div style={{ fontSize: 30, fontWeight: 500, color: '#93c5fd', marginTop: 20 }}>AI-Powered Merchant Cash Advance Underwriting</div>
        <div style={{ marginTop: 34, height: 4, width: 220, background: ACCENT, borderRadius: 4, marginLeft: 'auto', marginRight: 'auto' }} />
      </div>
    </AbsoluteFill>
  );
};

const Caption = ({ text, durationInFrames }) => {
  const frame = useCurrentFrame();
  const inOp = interpolate(frame, [0, 12], [0, 1], { extrapolateRight: 'clamp' });
  const outOp = interpolate(frame, [durationInFrames - 12, durationInFrames], [1, 0], { extrapolateLeft: 'clamp' });
  const y = interpolate(frame, [0, 12], [24, 0], { extrapolateRight: 'clamp' });
  return (
    <AbsoluteFill style={{ justifyContent: 'flex-end', alignItems: 'center', fontFamily: FONT, paddingBottom: 48 }}>
      <div style={{
        opacity: inOp * outOp, transform: `translateY(${y}px)`,
        background: 'rgba(10,22,40,0.92)', color: '#fff', padding: '14px 28px', borderRadius: 12,
        borderLeft: `5px solid ${ACCENT}`, fontSize: 27, fontWeight: 600, maxWidth: 1120,
        boxShadow: '0 12px 40px rgba(0,0,0,0.45)', backdropFilter: 'blur(2px)',
      }}>{text}</div>
    </AbsoluteFill>
  );
};

const Watermark = () => (
  <AbsoluteFill style={{ fontFamily: FONT, pointerEvents: 'none' }}>
    <div style={{ position: 'absolute', top: 22, right: 26, background: 'rgba(10,22,40,0.88)', color: '#fff', padding: '8px 16px', borderRadius: 999, fontSize: 18, fontWeight: 700, letterSpacing: 0.3, boxShadow: '0 6px 20px rgba(0,0,0,0.35)' }}>
      <span style={{ color: ACCENT }}>◆</span> MCA-Tracker
    </div>
  </AbsoluteFill>
);

const Outro = ({ durationInFrames }) => {
  const frame = useCurrentFrame();
  const op = interpolate(frame, [0, 18], [0, 1], { extrapolateRight: 'clamp' });
  return (
    <AbsoluteFill style={{ background: `radial-gradient(120% 120% at 50% 100%, ${NAVY700} 0%, ${NAVY} 60%)`, justifyContent: 'center', alignItems: 'center', fontFamily: FONT }}>
      <div style={{ opacity: op, textAlign: 'center', color: '#fff' }}>
        <div style={{ fontSize: 64, fontWeight: 800, letterSpacing: -1.5 }}>Re-wire, don’t rebuild.</div>
        <div style={{ fontSize: 28, color: '#93c5fd', marginTop: 18, maxWidth: 900 }}>The AI underwriting engine is CRM-agnostic — ready for the Salesforce transition.</div>
        <div style={{ marginTop: 32, color: ACCENT, fontSize: 20, fontWeight: 700, letterSpacing: 4, textTransform: 'uppercase' }}>MCA-Tracker · built for AFG</div>
      </div>
    </AbsoluteFill>
  );
};

export const Demo = ({ intro, video, outro }) => {

            <Caption text={c.text} durationInFrames={Math.round(c.d * fps)} />
          </Sequence>
        ))}
      </Sequence>

      <Sequence from={intro + video} durationInFrames={outro}><Outro durationInFrames={outro} /></Sequence>
    </AbsoluteFill>

};
```
